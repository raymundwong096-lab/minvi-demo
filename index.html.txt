    <script>
        // ============================================================
        // 1. BACKGROUND HEARTS
        // ============================================================
        (function createBgHearts() {
            const container = document.getElementById('bgHearts');
            const symbols = ['♥', '❤', '🩷', '🌸', '🌹'];
            for (let i = 0; i < 24; i++) {
                const span = document.createElement('span');
                span.textContent = symbols[i % symbols.length];
                span.style.left = Math.random() * 100 + '%';
                span.style.fontSize = (1.2 + Math.random() * 1.8) + 'rem';
                span.style.animationDuration = (18 + Math.random() * 32) + 's';
                span.style.animationDelay = (Math.random() * 30) + 's';
                container.appendChild(span);
            }
        })();

        // ============================================================
        // 2. SLIDE MANAGEMENT
        // ============================================================
        const slides = document.querySelectorAll('.slide');
        const dots = document.querySelectorAll('.dot');
        let currentSlide = 0;
        let isTransitioning = false;

        function goToSlide(index, skipEnvelope = false) {
            if (isTransitioning || index === currentSlide) return;
            if (index < 0 || index >= slides.length) return;
            // if trying to go to story before envelope is opened, ignore
            if (index === 1 && !skipEnvelope && !letterVisible) {
                // but if we're coming from envelope click, let it through
                return;
            }

            isTransitioning = true;

            // remove active from all slides
            slides.forEach((s, i) => {
                s.classList.remove('active', 'exit');
                if (i !== index) {
                    s.classList.add('exit');
                }
            });

            // activate target
            slides[index].classList.add('active');
            slides[index].classList.remove('exit');

            // update dots
            dots.forEach((d, i) => {
                d.classList.toggle('active', i === index);
            });

            currentSlide = index;

            // if going to story, ensure we scroll to top
            if (index === 1) {
                const scrollEl = document.getElementById('storyScroll');
                if (scrollEl) scrollEl.scrollTop = 0;
            }

            setTimeout(() => {
                isTransitioning = false;
            }, 700);
        }

        // dot clicks
        dots.forEach((dot) => {
            dot.addEventListener('click', () => {
                const idx = parseInt(dot.dataset.slide);
                if (idx === 1 && !letterVisible) {
                    // if envelope not opened, open it first
                    openEnvelope();
                    setTimeout(() => { goToSlide(1, true); }, 800);
                    return;
                }
                goToSlide(idx, true);
            });
        });

        // ============================================================
        // 3. ENVELOPE LOGIC
        // ============================================================
        const envelope = document.getElementById('envelope');
        const envelopeWrapper = document.getElementById('envelopeWrapper');
        const letterContent = document.getElementById('letterContent');
        const openStoryBtn = document.getElementById('openStoryBtn');
        let letterVisible = false;

        function openEnvelope() {
            if (letterVisible) return;
            letterVisible = true;
            envelope.classList.add('open');
            // show letter
            setTimeout(() => {
                letterContent.classList.add('show');
                // update envelope wrapper to not be clickable
                envelopeWrapper.style.cursor = 'default';
            }, 400);
        }

        // click envelope to open
        envelopeWrapper.addEventListener('click', (e) => {
            e.stopPropagation();
            if (!letterVisible) {
                openEnvelope();
            }
        });

        // "View Our Story" button
        openStoryBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            if (!letterVisible) {
                openEnvelope();
                setTimeout(() => { goToSlide(1, true); }, 800);
            } else {
                goToSlide(1, true);
            }
        });

        // ============================================================
        // 4. RSVP BUTTON
        // ============================================================
        document.getElementById('rsvpBtn').addEventListener('click', () => {
            const messages = [
                '💖 You just made my heart skip a beat!',
                '🌸 I knew you\'d say yes! See you soon, love!',
                '❤️ I\'m counting the seconds until I see you!',
                '🌹 You\'re the best thing that ever happened to me.'
            ];
            const msg = messages[Math.floor(Math.random() * messages.length)];
            // replace button with celebration
            const btn = document.getElementById('rsvpBtn');
            btn.textContent = '🎉 ' + msg;
            btn.style.background = 'linear-gradient(135deg, #2d8a4e, #1a5c33)';
            btn.style.boxShadow = '0 6px 30px rgba(45, 138, 78, 0.4)';
            btn.style.cursor = 'default';
            btn.disabled = true;

            // confetti effect (simple)
            for (let i = 0; i < 18; i++) {
                setTimeout(() => {
                    const el = document.createElement('span');
                    el.textContent = ['❤️', '🌹', '✨', '💖', '🌸', '🎉', '💕', '🌺'][Math.floor(Math.random() * 8)];
                    el.style.position = 'fixed';
                    el.style.left = (20 + Math.random() * 60) + '%';
                    el.style.top = '-10%';
                    el.style.fontSize = (1.8 + Math.random() * 2.4) + 'rem';
                    el.style.zIndex = '999';
                    el.style.pointerEvents = 'none';
                    el.style.transition = 'all 3s cubic-bezier(0.2, 0.8, 0.3, 1)';
                    el.style.opacity = '1';
                    document.body.appendChild(el);
                    requestAnimationFrame(() => {
                        el.style.transform = `translateY(${70 + Math.random() * 40}vh) rotate(${360 + Math.random() * 720}deg) scale(${0.6 + Math.random() * 0.8})`;
                        el.style.opacity = '0';
                    });
                    setTimeout(() => el.remove(), 3500);
                }, i * 120);
            }
        });

        // ============================================================
        // 5. KEYBOARD NAV (optional)
        // ============================================================
        document.addEventListener('keydown', (e) => {
            if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
                const next = Math.min(currentSlide + 1, slides.length - 1);
                if (next === 1 && !letterVisible) {
                    openEnvelope();
                    setTimeout(() => { goToSlide(1, true); }, 800);
                } else {
                    goToSlide(next, true);
                }
            } else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
                const prev = Math.max(currentSlide - 1, 0);
                goToSlide(prev, true);
            }
        });

        // ============================================================
        // 6. AUTO-OPEN ENVELOPE ON FIRST VISIT (gentle)
        // ============================================================
        // wait a beat then show a subtle pulse on envelope
        setTimeout(() => {
            if (!letterVisible) {
                envelopeWrapper.style.animation = 'pulseHint 2s ease-in-out 2';
                setTimeout(() => {
                    envelopeWrapper.style.animation = '';
                }, 4000);
            }
        }, 1200);

        // ============================================================
        // 7. ensure story scroll resets when entering
        // ============================================================
        const storyScroll = document.getElementById('storyScroll');
        if (storyScroll) {
            const observer = new MutationObserver(() => {
                if (slides[1].classList.contains('active')) {
                    storyScroll.scrollTop = 0;
                }
            });
            observer.observe(slides[1], { attributes: true, attributeFilter: ['class'] });
        }

        console.log('💌 Happy 1st Monthsary, love! — asyouwish');
